let selected = new Set();
function renderIngs(){
const c=document.getElementById('ingredients');
c.innerHTML=ings.map(i=>`
<div onclick="toggle('${i.id}')" id="ing-${i.id}" class="bg-white rounded-2xl p-3 shadow-sm border-2 border-transparent cursor-pointer relative">
<span class="absolute top-2 left-2 bg-orange-100 text-orange-600 text-[10px] px-2 py-1 rounded-full">${i.qty}</span>
<div class="text-3xl text-center my-2">${i.emoji}</div>
<div class="text-center font-bold text-sm">${i.ar}<br><span class="text-xs font-normal text-gray-400">${i.en}</span></div>
</div>`).join('');
}
function toggle(id){
if(selected.has(id))selected.delete(id);else selected.add(id);
document.getElementById('ing-'+id).classList.toggle('border-[#FF8C42]');
document.getElementById('ing-'+id).classList.toggle('bg-orange-50');
document.getElementById('count').innerText=`${selected.size} مكون مختار • ${selected.size} Selected`;
}
function suggest(){
const list = recipes.filter(r=>r.needs.some(n=>selected.has(n))).sort((a,b)=>b.needs.filter(n=>selected.has(n)).length - a.needs.filter(n=>selected.has(n)).length);
if(list.length==0){document.getElementById('recipes').innerHTML='<div class="text-center p-4 bg-white rounded-xl">اختارلك كم مكون فوك حتى اقترحلك • Select ingredients first</div>';return;}
document.getElementById('recipes').innerHTML=list.map(r=>{
const match=r.needs.filter(n=>selected.has(n)).length;
return `<div class="bg-white rounded-2xl p-4 shadow">
<h3 class="font-bold">${r.title} - ${r.en}</h3>
<p class="text-xs text-gray-400 my-1">${r.time} • ${match}/${r.needs.length} مكون متوفر</p>
<p class="text-sm mt-2">${r.desc}</p>
<button class="mt-3 w-full bg-[#FFFBF5] border border-orange-200 text-orange-600 py-2 rounded-xl text-sm">شوف الطريقة كاملة • View Recipe</button>
</div>`}).join('');
}
renderIngs();suggest();
